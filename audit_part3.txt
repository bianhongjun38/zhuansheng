 595:     private void addItemBar(LinearLayout parent) {
 596:         org.json.JSONArray ids = store.items();
 597:         if (ids.length() == 0) return;
 598:         java.util.Set<String> seen = new java.util.HashSet<>();
 599:         for (int i = 0; i < ids.length(); i++) {
 600:             final String id = ids.optString(i);
 601:             if (id == null || id.isEmpty() || !seen.add(id)) continue;
 602:             GameData.Item found = null;
 603:             for (GameData.Item it : data.items) {
 604:                 if (it.id.equals(id)) { found = it; break; }
 605:             }
 606:             if (found == null) continue;
 607:             final GameData.Item item = found;

 795:     /**
 796:      * 按题型挑一句合适的中文台词。c 是这轮打出去的牌。
 797:      *
 798:      * ── 例句题说的是「整句的翻译」，不是单个单词 ────────────────
 799:      *
 800:      * 旧版无论什么题型都只念单词的第一个义项，例句题也念。
 801:      * 但例句题考的本来就是「你能不能读懂这一整句」——
 802:      * 玩家读着 The captain gave the order to ___ ship. 选对了 abandon，
 803:      * 怪物却只回「放弃，抛弃」两个字，等于把最大的学习收益（整句释义）
 804:      * 白白扔掉，而且答对一句长句却只换来两个字的回应，反馈严重不足。
 805:      *
 806:      * 现在例句题走 exCnFirst（数据侧已经切好第一句，见 GCard），
 807:      * 把整句中文念出来；没有译文数据时才退回单词义。
 808:      * 其余题型仍然念单词义 —— 那些题型本来就没给句子。
 809:      */
 810:     private void enemySpeakFor(GCard c, String explain) {
 811:         if (c == null) { enemySpeak(null); return; }
 812: 
 813:         String sense = c.isWord() ? GQuiz.firstSensePublic(c.cn) : "";
 814:         if (sense == null) sense = "";
 815:         sense = sense.trim();
 816: 
 817:         // 例句题优先念整句译文
 818:         String sentence = c.exCnFirst == null ? "" : c.exCnFirst.trim();
 819:         if (pendingQLastMode == 1 && !sentence.isEmpty()) {
 820:             // 译文可能很长（有些例句译文三十多个字），台词框只有一行，
 821:             // 超出会被截断成省略号。这里按 44 字上限收一收，保住完整语义。
 822:             enemySpeak("「" + shorten(sentence, 44) + "」……这一句我本来没打算让你读懂。");
 823:             return;
 824:         }
 825: 
 826:         if (sense.isEmpty()) { enemySpeak(null); return; }
 827:         enemySpeak("「" + sense + "」……你居然真懂这个词。");
 828:     }

 830:     /**
 831:      * 超长文本截到 max 字以内，末尾补省略号。

 833:      * 中文按字算，一个汉字算一个 —— 台词框宽度固定，不截会直接溢出。
 834:      */
 835:     private static String shorten(String s, int max) {
 836:         if (s == null) return "";
 837:         String t = s.trim();
 838:         if (t.length() <= max) return t;
 839:         return t.substring(0, max) + "…";
 840:     }

 867:     private void renderQuiz() {
 868:         optRow.removeAllViews();
 869: 
 870:         // 例句题句子长，字号要小一点，否则一个屏幕放不下
 871:         boolean isExample = pendingQ.mode == 1;
 872:         TextView q = tv(pendingQ.prompt, isExample ? 15 : 20, C_FG);
 873:         q.setTypeface(Typeface.DEFAULT_BOLD);
 874:         q.setGravity(Gravity.CENTER);
 875:         q.setPadding(dp(4), dp(2), dp(4), 0);
 876:         if (isExample) q.setLineSpacing(dp(3), 1f);
 877:         optRow.addView(q);
 878: 
 879:         if (!pendingQ.sub.isEmpty()) {
 880:             TextView s = tv(pendingQ.sub, 11.5f, C_FG3);
 881:             s.setGravity(Gravity.CENTER);
 882:             s.setPadding(dp(4), dp(3), dp(4), dp(8));
 883:             optRow.addView(s);
 884:         }
 885: 
 886:         if (pendingQ.options != null) {
 887:             // 释义辨析题：选项是屏幕上的按钮（干扰项都不是手牌）
 888:             //
 889:             // lucid（精英掉落）：每题排除一个错误选项。
 890:             // 在界面层划掉而不是改题库数据 —— 题库是共用的，改一处会污染
 891:             // 后续所有题目；这里只在渲染时决定「这一题这个选项不可点」。
 892:             int cut = -1;
 893:             if (battle != null && battle.spLucid && pendingQ.options.length > 2) {
 894:                 java.util.Random lr = new java.util.Random(pendingQ.prompt.hashCode());
 895:                 for (int tries = 0; tries < 8; tries++) {
 896:                     int k = lr.nextInt(pendingQ.options.length);
 897:                     if (k != pendingQ.answer) { cut = k; break; }
 898:                 }
 899:             }
 900:             for (int i = 0; i < pendingQ.options.length; i++) {
 901:                 final int k = i;
 902:                 TextView ob = optBtn(pendingQ.options[i], () -> answer(k));
 903:                 if (k == cut) {
 904:                     ob.setText("✗ " + pendingQ.options[i]);
 905:                     ob.setAlpha(0.35f);
 906:                     ob.setPaintFlags(ob.getPaintFlags()
 907:                             | android.graphics.Paint.STRIKE_THRU_TEXT_FLAG);
 908:                     ob.setOnClickListener(null);   // 划掉的就是不能点
 909:                     ob.setClickable(false);
 910:                 }
 911:                 optRow.addView(ob);
 912:             }
 913:         } else if (pendingQ.mode == 4) {
 914:             // 僻义过载走的是选项制，答案不在手上 —— 提示语不能写「点牌」，
 915:             // 否则玩家会去点手牌，而那正是这题要避免的作弊路径。
 916:             TextView tip = tv("▾ 选出这个词的生僻义项", 12, C_PURPLE);
 917:             tip.setGravity(Gravity.CENTER);
 918:             tip.setPadding(0, dp(6), 0, dp(2));
 919:             optRow.addView(tip);
 920:         } else {
 921:             // 其余题型：答案在手上，点牌作答
 922:             TextView tip = tv("▾ 点你手上对应的那张牌", 12, C_TEAL);
 923:             tip.setGravity(Gravity.CENTER);
 924:             tip.setPadding(0, dp(6), 0, dp(2));
 925:             optRow.addView(tip);
 926:         }
 927: 
 928:         // ── 僻义过载入口 ──────────────────────────────────
 929:         //
 930:         // 只有当前题目的目标牌有第二个义项时才出现。点它会把题面换成
 931:         // 生僻义项重新出题，答对 2.2 倍真实伤害、答错自伤 4 点。
 932:         // 放在最底下、用醒目色，是因为它是个「加注」操作，
 933:         // 不该和正常作答混在一起被误触。
 934:         if (pendingQ.card != null && pendingQ.card.canOverload()
 935:                 && pendingQ.mode != 4) {
 936:             // 【绝不能写 pendingQ.card.en】。
 937:             // 这一题问的是「中文 → 从手牌里挑出那个词」，答案就是这张牌的单词。
 938:             // 之前按钮写成「🔥 僻义过载 · <单词>」，等于把答案直接印在题面上，
 939:             // 玩家不用辨识，照着按钮上的拼写去点牌就行 —— 和手牌上
 940:             //「◆僻义2」标记泄露答案是同一种错误。
 941:             // 所以这里只说加注规则，不透露是哪张牌。
 942:             optRow.addView(smallBtn("🔥 僻义过载（答对 2.2×，答错自伤 "

 948:     /**
 949:      * 把当前题目换成僻义题。
 950:      *
 951:      * 目标牌仍然是玩家按过载时选中的那张（他手上有它、答对就打它），
 952:      * 但**题面改成给单词选释义**：四选一挑它的僻义。
 953:      *
 954:      * 为什么不能沿用「给中文选牌」：手牌上印着「◆僻义2」，
 955:      * 玩家点哪张都看得见那个标记 —— 题面问「哪个是它的僻义」时，
 956:      * 带 ◆ 的牌本身就把答案指出来了。改成选释义之后，标记不再是提示，
 957:      * 因为这时要答的是「哪个意思」，而不是「哪张牌」。
 958:      */
 959:     private void startOverload() {
 960:         if (battle == null || pendingQ == null) return;
 961:         GCard target = pendingQ.card;
 962:         GQuiz.Q oq = quiz.makeOverload(target);
 963:         if (oq == null) {
 964:             addLog("这张牌没有可考的僻义", C_AMBER);
 965:             return;
 966:         }
 967:         pendingQ = oq;
 968:         // 这里同样不能带 target.en —— flash/addLog 会立刻把答案显示出来。
 969:         flash = "🔥 僻义过载：改考生僻义项";
 970:         addLog(flash, C_PURPLE);
 971:         render();
 972:     }
 973: 
 974:     /**
 975:      * 推进一回合，先把敌人这次的攻击"演"出来再刷界面。
 976:      *
 977:      * 原实现直接 battle.endTurn(false); render()，敌人的攻击只体现在
 978:      * 血量数字瞬间变小上，怪物一个动作都没有 —— 玩家看不出是挨打了。
 979:      * 这里先算一次 endTurn，比较前后血量：掉血了就播 attack，
 980:      * 让动作和伤害数字对得上。
 981:      */
 982:     private void advanceTurn() {
 983:         if (battle == null || battle.over()) return;
 984:         // 结束回合时若还有题没答，必须先清掉：否则 pendingQ 会带进下一回合，
 985:         // 新回合的牌对不上旧题目，玩家怎么点都是错的。
 986:         if (pendingQ != null) {
 987:             pendingQ = null;
 988:             pending = null;
 989:         }
 990:         int hpBefore = battle.playerHp;
 991:         battle.endTurn(false);
 992: 
 993:         boolean hurt = battle.playerHp < hpBefore;
 994:         if (battle.phase == GBattle.PHASE_LOSE) {
 995:             render();
 996:             return;
 997:         }
 998:         if (hurt) {
 999:             final int dmg = hpBefore - battle.playerHp;
1000:             flash = "⚔ " + (battle.enemy == null ? "敌人" : battle.enemy.name)
1001:                     + " 攻击　你受到 " + dmg + " 伤害";
1002:             addLog(flash, C_RED);
1003:             // 演出交给 enemyStrike：后撤→前冲→命中(闪红+抖动+残影+飘字)→回位
1004:             enemyStrike(new Runnable() {
1005:                 @Override public void run() { playerHurtFx(dmg); }
1006:             });
1007:         } else {
1008:             addLog("── 你结束了回合（敌人还在蓄力）", C_FG3);
1009:             sprite.play("move");
1010:         }
1011:         render();
1012:     }

1014:     /**
1015:      * 选项制题（题型 3 释义辨析）：点按钮作答。
1016:      *
1017:      * 答错时把**正确答案那张牌**弃掉 —— 因为选项不是实体牌，
1018:      * 没法「弃掉你点错的那个词」。题考的是这张牌，认不出来就该失去它。
1019:      */
1020:     private void answer(int pick) {
1021:         if (pendingQ == null || pendingQ.options == null) return;
1022:         GCard target = pendingQ.card;
1023: 
1024:         // 僻义过载（mode 4）也是选项题，但判对错的方式不同：
1025:         // 它选的是「哪个释义」，不是「哪个单词」，所以直接比 answer 下标，
1026:         // 不能走下面那个和 target.en 比字符串的分支。
1027:         if (pendingQ.mode == 4) {
1028:             // 不论对错都要记住义项下标：答对要用它结算 2.2 倍伤害，
1029:             // 答错要用它打出「你答的是什么、正确答案是什么」的说明。
1030:             overloadIndex = pendingQ.senseIndex;
1031: 
1032:             // ── 同义容差：选中的选项只要和正确答案「是同一个意思」就算对 ──
1033:             //
1034:             // 原来是纯下标比较（pick == pendingQ.answer）。出题侧已经尽量
1035:             // 保证选项互不同义，但中文释义本来就是一堆近义写法
1036:             // （收费/费用、进步/进展、相等的/等量），题库里同一概念
1037:             // 常有多种表述，光靠出题去重不可能穷尽。
1038:             //
1039:             // 一旦漏网，玩家选一个明明说得通的选项却被判错，
1040:             // 就是「生僻义回答常见意思也算错」那种冤枉感。
1041:             // 所以判题侧再兜一层：文字归一化后相等或互相包含即算对。
1042:             boolean right = pick == pendingQ.answer;
1043:             if (!right && pick >= 0 && pick < pendingQ.options.length
1044:                     && pendingQ.answer >= 0 && pendingQ.answer < pendingQ.options.length) {
1045:                 right = GQuiz.sameSensePublic(pendingQ.options[pick],
1046:                                                pendingQ.options[pendingQ.answer]);
1047:             }
1048:             resolve(target, right);
1049:             return;
1050:         }
1051: 
1052:         String chose = (pick >= 0 && pick < pendingQ.options.length)
1053:                 ? pendingQ.options[pick] : "";
1054:         boolean right = chose.equalsIgnoreCase(target == null ? "" : target.en);
1055:         resolve(target, right);
1056:     }

1568:      *

1570:      */
1571:     private void recordBattle(boolean win) {
1572:         try {
1573:             if (bpRev == null || bpRev.isEmpty()) return;
1574:             GStore.Battle b = new GStore.Battle();
1575:             b.day = com.zsb.plan.StateLogic.todayIso();
1576:             // ── 爬塔进度用「总层数」而不是 layer() ──────────────
1577:             //
1578:             // 这是排行榜一直没数据的真正原因。store.layer() 只在
1579:             // 一整张地图走到尽头时才 +1（见 GModeView.advanceStep），
1580:             // 而一张图有好几层 —— 大多数玩家整局都停在第 0 层。
1581:             // 于是一路爬到第 8 层的人，榜上显示的还是「第 1 层」，
1582:             // 看起来就像没同步。
1583:             //
1584:             // 真实爬塔进度 = layer() * 每层行数 + curRow()。
1585:             // 这样既包括跨地图的层数，也包括地图内已经走过的行数，
1586:             // 和玩家在界面上看到的 Lv. 对得上。
1587:             // 层数口径统一走 store.floorIndex()，不再在这里重算。
1588:             // 之前战斗强度、商店价格、顶栏、排行榜各算一套，
1589:             // 同一个「第 3 层」在不同地方指的东西不一样。
1590:             b.floor = store.floorIndex();
1591:             b.node = store.layer();   // 第几张地图 —— 排行榜也要能看到
1592:             b.win = win;
1593:             b.gold = Math.max(0, store.gold() - bpGold0);
1594:             b.right = Math.max(0, store.right() - bpRight0);
1595:             b.wrong = Math.max(0, store.wrong() - bpWrong0);
1596:             b.ms = Math.max(0, System.currentTimeMillis() - bpT0);
1597:             b.started = bpT0;
1598:             b.rev = bpRev;
1599:             // 一题都没答就结束的战斗（全程「直接攻击」）也记。
1600:             //
1601:             // 原来这里要求 total()>0，理由是"服务端会拒收"。但站在玩家
1602:             // 角度看这是错的：他确实爬到了第几层、确实打完了这场，
1603:             // 排行榜却完全没有他 —— 表现就是"我的爬塔记录没同步"。
1604:             // 战败的那一场更冤：输了本来就掉层，干脆连存在感都没有。
1605:             //
1606:             // 现在一律记录；服务端若因题目数太少而忽略，那是它的判断，
1607:             // 客户端不该提前替它扔掉玩家的成绩。
1608:             store.addBattle(b);
1609:         } catch (Exception ignored) {
1610:             // 记不上不能影响玩家继续玩
1611:         }
1612:         // 不再在战斗结束时自动上报 —— 排行榜改成玩家主动上传（见 BoardView.doUpload）。
1613:         // 原来每打一场就传一条，服务端按条累加，于是榜上是「一堆散记录
1614:         // 加起来的数」，玩家看到的金币/层数和自己界面对不上。
1615:     }

 265:     private static class Resp {
 266:         int code;
 267:         String body = "";

 270:     /**
 271:      * 发一次请求并把响应体读全。
 272:      * 网络层异常不在这里翻译，统一抛给上层用 netMsg() 转中文。
 273:      */
 274:     private Resp send(String method, String url, String body) throws IOException {
 275:         HttpURLConnection c = null;
 276:         try {
 277:             c = (HttpURLConnection) new URL(url).openConnection();
 278:             c.setRequestMethod(method);
 279:             c.setConnectTimeout(TIMEOUT);
 280:             c.setReadTimeout(TIMEOUT);
 281:             c.setRequestProperty("Authorization", "Bearer " + token);
 282:             c.setRequestProperty("Accept", "application/vnd.github+json");
 283:             c.setRequestProperty("User-Agent", "zsb-app");
 284:             c.setRequestProperty("X-GitHub-Api-Version", "2022-11-28");
 285:             c.setUseCaches(false);
 286: 
 287:             if (body != null) {
 288:                 c.setDoOutput(true);
 289:                 c.setRequestProperty("Content-Type", "application/json; charset=utf-8");
 290:                 OutputStream os = c.getOutputStream();
 291:                 try {
 292:                     os.write(body.getBytes("UTF-8"));
 293:                     os.flush();
 294:                 } finally {
 295:                     os.close();
 296:                 }
 297:             }
 298: 
 299:             Resp r = new Resp();
 300:             r.code = c.getResponseCode();
 301:             InputStream in = null;
 302:             try {
 303:                 if (r.code >= 200 && r.code < 300) in = c.getInputStream();
 304:                 else in = c.getErrorStream();
 305:                 r.body = in == null ? "" : readAll(in);
 306:             } catch (IOException ignored) {
 307:                 // 错误响应偶发没有 body，保持空串即可
 308:             } finally {
 309:                 if (in != null) try { in.close(); } catch (IOException ignored) { }
 310:             }
 311:             return r;
 312:         } finally {
 313:             if (c != null) c.disconnect();
 314:         }
 315:     }

 317:     private static String readAll(InputStream in) throws IOException {
 318:         ByteArrayOutputStream bo = new ByteArrayOutputStream();
 319:         byte[] buf = new byte[8192];
 320:         int n;
 321:         while ((n = in.read(buf)) > 0) bo.write(buf, 0, n);
 322:         return new String(bo.toByteArray(), "UTF-8");
 323:     }

 337:     /** 把 HTTP 状态码翻成人话，带上 GitHub 自己的 message 便于定位。 */
 338:     static String httpMsg(int code, String body) {
 339:         String detail = "";
 340:         if (body != null && !body.isEmpty()) {
 341:             try {
 342:                 JSONObject o = new JSONObject(body);
 343:                 if (o.has("message") && !o.isNull("message")) detail = o.optString("message", "");
 344:             } catch (Throwable ignored) {
 345:                 // 不是 JSON（多半是网关返回的 HTML），忽略
 346:             }
 347:             if (detail.length() > 120) detail = detail.substring(0, 120) + "…";
 348:         }
 349: 
 350:         String base;
 351:         if (code == 401) {
 352:             base = "Token 无效或已过期（401），去 GitHub 设置里重新生成一个带 gist 权限的 token";
 353:         } else if (code == 403) {
 354:             base = "GitHub 拒绝了请求（403）：token 缺 gist 权限，或请求太频繁被限流，稍再试";
 355:         } else if (code == 404) {
 356:             base = "找不到这个 Gist（404），检查 Gist ID 是否贴错";
 357:         } else if (code == 422) {
 358:             base = "云端不接受这次提交（422），进度内容可能过大或格式不对";
 359:         } else if (code >= 500) {
 360:             base = "GitHub 服务器出错（" + code + "），过一会儿再试";
 361:         } else {
 362:             base = "请求失败（HTTP " + code + "）";
 363:         }
 364:         return detail.isEmpty() ? base : base + "；" + detail;
 365:     }

 369:         if (t instanceof SocketTimeoutException) {
 370:             return "连接超时（15 秒），网络不稳或 GitHub 暂时连不上，换个网络再试";
 371:         }

 377:         }

 397:     /**
 398:      * 合并两端进度 JSON，返回合并后的 JSON 字符串。永不抛异常。
 399:      *
 400:      * 规则（按「宁可多记完成，不可丢进度」的原则）：
 401:      *   ① done / completed / star —— 勾选类，**并集**。任一端标记完成即视为完成。
 402:      *      这是整个策略的核心：设备 A 勾了、设备 B 没勾，合并后是勾上的。
 403:      *   ② hard / timing —— 数值类，**取较大值**。错题次数 n 取大；
 404:      *      计时 ms 也取大（严格说累计该相加，但双端都在累加同一段时间，
 405:      *      相加会把时长翻倍，取大更接近真实投入）。
 406:      *   ③ notes —— 文本类，哪端非空用哪端；两端都有内容且不同，**用本地**
 407:      *      （本地是用户刚敲的，远端多半是旧稿）。这条是唯一偏向本地的规则。
 408:      *   ④ wrong（错题本，若有）—— 按题目 id 对齐：n 取大、ts 取新、
 409:      *      学科/题干哪端非空用哪端。
 410:      *   ⑤ 其它未知字段 —— 本地有就留本地，本地没有再取远端，不丢字段。
 411:      *   ⑥ updatedAt 统一改写为合并时刻。
 412:      *
 413:      * 输入既可以是 StateLogic.exportJson() 那种 {_app,_version,state:{...}} 外层包装，
 414:      * 也可以是 StateLogic.serialize() 那种裸状态对象；两端形态不一致也能合。
 415:      * 输出形态跟随输入：任一端有包装就输出带包装的结构（跟 exportJson 一致）。
 416:      * 任一端为 null/空/坏 JSON 时，等价于只取另一端，不会返回 null。
 417:      */
 418:     public static String merge(String localJson, String remoteJson) {
 419:         JSONObject lo = parseObj(localJson);
 420:         JSONObject ro = parseObj(remoteJson);
 421: 
 422:         JSONObject ls = stateOf(lo);
 423:         JSONObject rs = stateOf(ro);
 424:         boolean wrapped = (lo != null && lo.optJSONObject("state") != null)
 425:                 || (ro != null && ro.optJSONObject("state") != null);
 426: 
 427:         JSONObject out;
 428:         if (ls == null && rs == null) {
 429:             out = new JSONObject();
 430:         } else if (ls == null) {
 431:             out = copy(rs);
 432:         } else if (rs == null) {
 433:             out = copy(ls);
 434:         } else {
 435:             out = mergeState(ls, rs);
 436:         }
 437: 
 438:         try {
 439:             if (!wrapped) return out.toString();
 440:             JSONObject root = new JSONObject();
 441:             root.put("_app", "zsb-plan");
 442:             root.put("_version", 1);
 443:             root.put("_exportedAt", nowIso());
 444:             root.put("state", out);
 445:             return root.toString();
 446:         } catch (Throwable t) {
 447:             return out.toString();
 448:         }
 449:     }

 451:     /** 合并两个裸状态对象。 */
 452:     private static JSONObject mergeState(JSONObject a, JSONObject b) {
 453:         JSONObject out = new JSONObject();
 454: 
 455:         // ① 勾选类：并集
 456:         unionBool(out, "done", a, b);
 457:         unionBool(out, "completed", a, b);
 458:         unionBool(out, "star", a, b);
 459: 
 460:         // ② 数值类：取大
 461:         maxNum(out, "hard", a, b);
 462:         maxNum(out, "timing", a, b);
 463: 
 464:         // ③ 文本类
 465:         mergeText(out, "notes", a, b);
 466: 
 467:         // ④ 错题本条目
 468:         mergeWrong(out, "wrong", a, b);
 469: 
 470:         // ⑤ 其余字段：本地优先，本地没有再取远端
 471:         copyRest(out, a, b);
 472:         copyRest(out, b, a);
 473: 
 474:         try {
 475:             out.put("updatedAt", nowIso());
 476:         } catch (Throwable ignored) {
 477:             // 放不进去就算了，不影响合并结果
 478:         }
 479:         return out;
 480:     }

 532:     /** 文本：非空优先，两端都非空且不同则本地（a）优先。 */
 533:     private static void mergeText(JSONObject out, String field, JSONObject a, JSONObject b) {
 534:         JSONObject ja = sub(a, field);
 535:         JSONObject jb = sub(b, field);
 536:         if (ja == null && jb == null) return;
 537:         JSONObject r = new JSONObject();
 538:         try {
 539:             if (jb != null) {
 540:                 for (Iterator<String> it = jb.keys(); it.hasNext(); ) {
 541:                     String k = it.next();
 542:                     String v = strOf(jb, k);
 543:                     if (v != null && !v.isEmpty()) r.put(k, v);
 544:                 }
 545:             }
 546:             if (ja != null) {
 547:                 for (Iterator<String> it = ja.keys(); it.hasNext(); ) {
 548:                     String k = it.next();
 549:                     String v = strOf(ja, k);
 550:                     if (v == null) continue;
 551:                     String old = strOf(r, k);
 552:                     // 本地非空就覆盖；本地为空但云端也为空，留本地那个空值
 553:                     if (!v.isEmpty() || old == null) r.put(k, v);
 554:                     else if (old.isEmpty()) r.put(k, v);
 555:                 }
 556:             }
 557:             out.put(field, r);
 558:         } catch (Throwable ignored) {
 559:         }
 560:     }

 562:     /**
 563:      * 错题本：按题目 id 对齐。
 564:      * n 取大（做错次数不因换设备而变少），ts 取新（最近一次做错时间），
 565:      * 学科/题干非空优先本地。
 566:      */
 567:     private static void mergeWrong(JSONObject out, String field, JSONObject a, JSONObject b) {
 568:         JSONObject ja = sub(a, field);
 569:         JSONObject jb = sub(b, field);
 570:         if (ja == null && jb == null) return;
 571:         JSONObject r = new JSONObject();
 572:         try {
 573:             java.util.HashSet<String> keys = new java.util.HashSet<>();
 574:             if (ja != null) for (Iterator<String> it = ja.keys(); it.hasNext(); ) keys.add(it.next());
 575:             if (jb != null) for (Iterator<String> it = jb.keys(); it.hasNext(); ) keys.add(it.next());
 576: 
 577:             for (String k : keys) {
 578:                 JSONObject ea = ja == null ? null : ja.optJSONObject(k);
 579:                 JSONObject eb = jb == null ? null : jb.optJSONObject(k);
 580:                 if (ea == null && eb == null) continue;
 581:                 JSONObject e = new JSONObject();
 582:                 // n / ts 取大
 583:                 long n = Math.max(ea == null ? 0 : ea.optLong("n", 0), eb == null ? 0 : eb.optLong("n", 0));
 584:                 long ts = Math.max(ea == null ? 0 : ea.optLong("ts", 0), eb == null ? 0 : eb.optLong("ts", 0));
 585:                 e.put("n", n <= 0 ? 1 : n);
 586:                 e.put("ts", ts);
 587:                 // 其余文本字段：本地非空优先
 588:                 e.put("s", firstNonEmpty(strOf(ea, "s"), strOf(eb, "s")));
 589:                 e.put("q", firstNonEmpty(strOf(ea, "q"), strOf(eb, "q")));
 590:                 r.put(k, e);
 591:             }
 592:             out.put(field, r);
 593:         } catch (Throwable ignored) {
 594:         }
 595:     }

 597:     /** 把 primary 里 out 还没有的字段搬过来（a 是 prefer 端）。 */
 598:     private static void copyRest(JSONObject out, JSONObject prefer, JSONObject other) {
 599:         if (prefer == null) return;

 601:             String k = it.next();

 607:             }
 608:         }
 609:         if (other == null) return;
 610:         for (Iterator<String> it = other.keys(); it.hasNext(); ) {
 611:             String k = it.next();
 612:             if (isKnown(k)) continue;
 613:             if (out.has(k)) continue;
 614:             try {
 615:                 out.put(k, other.opt(k));
 616:             } catch (Throwable ignored) {
 617:             }
 618:         }
 619:     }

 382:      * 实际做法：存档里把每层的节点类型存成 "0,2,1|2,4|..."（| 分层，, 分列），
 383:      * 连通性则存成 "0-1,0-2|1-0,1-2|..."（层内 h 连到下层 t）。
 384:      */
 385:     private GameData.RouteMap loadOrMakeMap() {
 386:         String raw = store.mapRaw();
 387:         if (raw == null || raw.isEmpty()) return null;
 388: 
 389:         try {
 390:             GameData.RouteMap m = new GameData.RouteMap();
 391:             String[] layerStrs = raw.split("\\|");
 392:             for (int r = 0; r < layerStrs.length; r++) {
 393:                 String ls = layerStrs[r];
 394:                 List<GameData.Node> layer = new ArrayList<>();
 395:                 if (!ls.isEmpty()) {
 396:                     String[] cols = ls.split(",");
 397:                     for (int c = 0; c < cols.length; c++) {
 398:                         int type = parseInt(cols[c]);
 399:                         layer.add(new GameData.Node(type, r, c));
 400:                     }
 401:                 }
 402:                 m.nodes.add(layer);
 403:             }
 404:             // 重建连通性：按 makeMap 同样的算法会不可复现，所以改用「按层节点数
 405:             // 均匀分配」的确定性规则 —— 与 makeMap 的"按比例映射"主边一致。
 406:             rebuildEdges(m);
 407:             return m;
 408:         } catch (Exception e) {
 409:             return null;
 410:         }
 411:     }
 412: 
 413:     /**
 414:      * 重建层间连边。规则与 makeMap 的主干边一致：上一层的第 i 个节点
 415:      * 按比例映射到下一层，再补齐所有下层节点的入边。
 416:      *
 417:      * 这样存档只需存「每层有几个节点、各是什么类型」，连通性完全确定，
 418:      * 读档不会走样。
 419:      */
 420:     private static void rebuildEdges(GameData.RouteMap m) {
 421:         for (int r = 0; r < m.rows() - 1; r++) {
 422:             List<GameData.Node> cur = m.row(r);
 423:             List<GameData.Node> nxt = m.row(r + 1);
 424:             int nc = nxt.size();
 425:             for (GameData.Node n : cur) n.next.clear();
 426:             for (int i = 0; i < cur.size(); i++) {
 427:                 int target = cur.size() == 1 ? 0
 428:                         : (int) Math.round((double) i * (nc - 1) / Math.max(1, cur.size() - 1));
 429:                 target = Math.max(0, Math.min(nc - 1, target));
 430:                 if (!cur.get(i).next.contains(target)) cur.get(i).next.add(target);
 431:             }
 432:             // 补齐入边
 433:             for (int j = 0; j < nc; j++) {
 434:                 boolean has = false;
 435:                 for (GameData.Node n : cur) if (n.next.contains(j)) has = true;
 436:                 if (has) continue;
 437:                 int src = cur.size() == 1 ? 0
 438:                         : Math.max(0, Math.min(cur.size() - 1,
 439:                             (int) Math.round((double) j * (cur.size() - 1) / Math.max(1, nc - 1))));
 440:                 cur.get(src).next.add(j);
 441:             }
 442:             for (GameData.Node n : cur) java.util.Collections.sort(n.next);
 443:         }
 444:     }

 446:     /** 把 GameData.RouteMap 序列化成存档字符串。 */
 447:     private static String mapToRaw(GameData.RouteMap m) {
 448:         StringBuilder sb = new StringBuilder();
 449:         for (int r = 0; r < m.rows(); r++) {
 450:             if (r > 0) sb.append("|");
 451:             List<GameData.Node> layer = m.row(r);
 452:             for (int c = 0; c < layer.size(); c++) {
 453:                 if (c > 0) sb.append(",");
 454:                 sb.append(layer.get(c).type);
 455:             }
 456:         }
 457:         return sb.toString();
 458:     }

 462:         String icon;

 470:         }
 471:

 476: 
 477:         if (isHere) {
 478:             // 当前位置：实心高亮
 479:             g.setColor(color);
 480:             g.setStroke(dp(2), 0xFFFFFFFF);
 481:         } else if (canGo) {
 482:             // 可去：描边发亮
 483:             g.setColor(C_BG3);
 484:             g.setStroke(dp(2), color);
 485:         } else if (passed) {
 486:             // 走过：压暗
 487:             g.setColor(0xFF191C22);
 488:             g.setStroke(dp(1), C_LINE);

 492:             g.setStroke(dp(1), C_LINE);
 493:         }
 494:         t.setBackground(g);
 495:         t.setAlpha(isHere || canGo || passed ? 1f : 0.45f);
 496:         return t;
