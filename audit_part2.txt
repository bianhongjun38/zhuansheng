 161:     /**
 162:      * 三个来源都查一遍，取**版本号最高**的那个，而不是"第一个能读到就用"。
 163:      *
 164:      * 为什么必须改成取最大值：
 165:      * 前两条通道读的是我们自己域名上的 update.json —— 一份要手动跑
 166:      * 发版pages.py 才会更新的静态文件。忘了跑，它就永远停在旧版本。
 167:      * 旧实现里它排在第一位且"读到即返回"，于是：
 168:      *   update.json 停在 1.2.3，GitHub 上已经是 1.4.1，
 169:      *   本机 1.4.1 → 10203 <= 10401 → 判定「已是最新版本」。
 170:      * 明明线上有新版、明明 GitHub 通道完全正常，却被一份过期静态文件
 171:      * 整个挡住。这不是"某个源挂了"，是取值策略错了。
 172:      *
 173:      * 现在任何一份过期都只是被忽略，不会压住更高版本。
 174:      *
 175:      * 同版本时优先保留带下载地址的那份：update.json 可能在 APK 超过
 176:      * Pages 单文件上限时把 apk 置为 null，那时要能补上 GitHub 的地址。
 177:      */
 178:     private Latest fetchLatestAny() {
 179:         // 探测顺序保持"自有域名优先"——只是不再提前返回，
 180:         // 三份都要拿到手才能比出最高的那个。
 181:         Latest[] all = {
 182:                 fetchManifest(PRIMARY_JSON),        // app.xxjcloudns.kdns.fr（Pages，主通道）
 183:                 fetchManifest(PAGES_JSON),          // zsb-update.pages.dev（Pages 兜底）
 184:                 fetchFromGithub(),                  // api.github.com
 185:         };
 186: 
 187:         Latest best = null;
 188:         for (Latest lt : all) {
 189:             if (lt == null || lt.code == 0) continue;
 190:             if (best == null) { best = lt; continue; }
 191:             if (lt.code > best.code) {
 192:                 best = lt;
 193:             } else if (lt.code == best.code
 194:                     && best.apkUrl == null && lt.apkUrl != null) {
 195:                 // 版本一样但这份能下载，补上地址
 196:                 best.apkUrl = lt.apkUrl;
 197:             }
 198:         }
 199:         return best;
 200:     }

 202:     /** 读一份 update.json 清单。 */
 203:     private Latest fetchManifest(String src) {
 204:         HttpURLConnection c = null;
 205:         try {
 206:             c = (HttpURLConnection) new URL(src).openConnection();
 207:             c.setConnectTimeout(8000);
 208:             c.setReadTimeout(10000);
 209:             c.setRequestProperty("User-Agent", "zsb-plan");
 210:             c.setRequestProperty("Cache-Control", "no-cache");
 211:             if (c.getResponseCode() != 200) return null;
 212:             String s = readAll(c.getInputStream());
 213:             JSONObject o = new JSONObject(s);
 214:             if (!o.optBoolean("ok", false)) return null;
 215:             String v = o.optString("versionName", "");
 216:             int code = o.optInt("versionInt", parseVer(v));
 217:             if (code == 0) return null;
 218:             Latest lt = new Latest();
 219:             lt.ver = v;
 220:             lt.code = code;
 221:             lt.notes = o.optString("notes", "");
 222:             JSONObject apk = o.optJSONObject("apk");
 223:             if (apk != null) lt.apkUrl = apk.optString("url", null);
 224:             return lt;
 225:         } catch (Exception e) {
 226:             return null;
 227:         } finally {
 228:             if (c != null) c.disconnect();
 229:         }
 230:     }

 278:     /** "1.1.0" / "v1.1.0" → 10100。解析不了返回 0。 */
 279:     static int parseVer(String s) {
 280:         if (s == null) return 0;
 281:         try {
 282:             String t = s.trim();
 283:             if (t.startsWith("v") || t.startsWith("V")) t = t.substring(1);
 284:             // 去掉可能的后缀，如 1.1.0-beta
 285:             int cut = t.length();
 286:             for (int i = 0; i < t.length(); i++) {
 287:                 char ch = t.charAt(i);
 288:                 if (ch != '.' && (ch < '0' || ch > '9')) { cut = i; break; }
 289:             }
 290:             t = t.substring(0, cut);
 291:             if (t.isEmpty()) return 0;
 292:             String[] p = t.split("\\.");
 293:             int maj = p.length > 0 ? Integer.parseInt(p[0]) : 0;
 294:             int min = p.length > 1 ? Integer.parseInt(p[1]) : 0;
 295:             int pat = p.length > 2 ? Integer.parseInt(p[2]) : 0;
 296:             return maj * 10000 + min * 100 + pat;
 297:         } catch (Exception e) {
 298:             return 0;
 299:         }
 300:     }

 348:     /**
 349:      * 取安装包的下载地址。
 350:      *
 351:      * **必须用 api.github.com 的资产端点，不能用 browser_download_url。**
 352:      *
 353:      * 2026-10-27 实测（不带代理，模拟手机处境）：
 354:      *   api.github.com                       HTTP 200   321ms
 355:      *   github.com                           HTTP 000   超时 12s   ← 卡死在这里
 356:      *   release-assets.githubusercontent.com HTTP 404   388ms（域名可达）
 357:      *
 358:      * browser_download_url 是 https://github.com/.../releases/download/...，
 359:      * 也就是走被阻断的那个域名 —— 版本能查到、但一按下载就永远转圈。
 360:      *
 361:      * api.github.com/repos/.../releases/assets/{id} 带 Accept: octet-stream
 362:      * 会 302 到 release-assets.githubusercontent.com（那个域名直连是通的），
 363:      * 实测完整下载 745 KB/s、1.4 秒拿 1 MB。
 364:      */
 365:     private String findApk(JSONObject rel) {
 366:         try {
 367:             JSONArray arr = rel.getJSONArray("assets");
 368:             for (int i = 0; i < arr.length(); i++) {
 369:                 JSONObject a = arr.getJSONObject(i);
 370:                 String name = a.optString("name", "");
 371:                 if (!name.endsWith(".apk")) continue;
 372:                 String api = a.optString("url", "");
 373:                 if (api.startsWith("https://api.github.com/")) {
 374:                     fallbackGithubUrl = api;
 375:                     return api;
 376:                 }
 377:                 fallbackGithubUrl = a.optString("browser_download_url", null);
 378:                 // 兜底：万一副本接口没给 url，退回老地址（可能很慢但不至于不可用）
 379:                 return a.optString("browser_download_url", null);
 380:             }
 381:         } catch (Exception ignored) {
 382:         }
 383:         return null;
 384:     }

 420:     /**
 421:      * 按顺序尝试多个下载源，任一成功即返回。
 422:      *
 423:      * 为什么要多源：本机 2026-10-28 实测——
 424:      *   github.com（release 资产，302 到 release-assets）完整 20.12 MB / 3.1s / 6.6 MB/s
 425:      *   app.xxjcloudns.kdns.fr（Pages，APK 3.65 MB / 2.9s / 1.3 MB/s）

 427:      * 单一源一旦抽风，用户就只能干等或失败；多源按序退避最稳。
 428:      *

 430:      * 20 MB 的包在国内掉一次线就全废，这个成本太高。
 431:      */
 432:     private void download(final java.util.List<String> urls) {
 433:         final java.util.List<String> sources = new java.util.ArrayList<>();
 434:         for (String u : urls) if (u != null && !sources.contains(u)) sources.add(u);
 435: 
 436:         final ProgressDialog pd = new ProgressDialog(act);
 437:         pd.setTitle("正在下载更新");
 438:         pd.setMessage("0%");
 439:         pd.setCancelable(false);
 440:         pd.show();
