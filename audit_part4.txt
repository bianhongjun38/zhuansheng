 898:     /**
 899:      * 一场战斗结束后推进地图。
 900:      *
 901:      * 这里修的是两个老 bug：
 902:      *   1. 原来是 raiseFloor(store.layer()) —— 参数就是自己，恒等于 0，
 903:      *      于是 bestFloor 永远是 0、顶栏永远显示 Lv.1。
 904:      *   2. 原来只做 node+1，没有任何翻层逻辑。node 走到路线最后一个节点后，
 905:      *      routeCard 只给 i == curNode 的节点绑定点击，而 i 最大就是 len-1，
 906:      *      于是节点越界、地图上一个能点的格子都没有 —— 玩家被卡死在层末。
 907:      */
 908:     private void advanceAfterBattle() {
 909:         // 宝箱怪打赢了：把开箱该拿的金币补上（战斗本身另有一份金币奖励，
 910:         // 见 GBattleView.showEnd）。两个都给，才对得起「开了箱子还打了一场」。
 911:         if (pendingChestGold > 0) {
 912:             store.addGold(pendingChestGold);
 913:             pendingChestGold = -1;
 914:         }
 915:         clearFightNode();        // 玩家确认过了，这一格才算真正走完
 916:         store.clearBattleNode();
 917:         advanceStep();
 918:         endBattle();
 919:     }
 920: 
 921:     /**
 922:      * 推进到地图上的下一步。
 923:      *
 924:      * 分叉地图没有「路线长度」概念了 —— 玩家站在 (row,col)，下一步是他
 925:      * 在那个节点上选的。但战斗结束后玩家还没选，所以流程是：
 926:      *   打完 → row+1（先落到下一层），下一层具体哪个格子等他点。
 927:      *
 928:      * 这里只负责「层数往上走 + 刷新 Lv. + 到底了翻新地图」；
 929:      * 具体站在下一层的哪一格，由玩家点哪个可达节点决定（enterNode 里 setPos）。
 930:      */
 931:     private void advanceStep() {
 932:         GameData.RouteMap map = loadOrMakeMap();
 933:         if (map == null) { renderMap(); return; }
 934: 
 935:         int row = store.curRow();
 936:         int nextRow = row + 1;
 937: 
 938:         if (nextRow >= map.rows()) {
 939:             // 走到底了：进新一层，整张地图重抽
 940:             store.setLayer(store.layer() + 1);
 941:             GameData.RouteMap nm = GameData.makeMap(rnd);
 942:             store.setMap(mapToRaw(nm));
 943:             store.setVisitedClear();
 944:             store.markVisited(0, 0);
 945:             store.setPos(0, -1);   // 新一层同样从岔路口开始
 946:         } else {
 947:             // 还没到底：位置先落在这一层，等玩家点可达格。
 948:             // 用 -1 表示"在层间过渡，尚未选定格子"。
 949:             store.setPos(nextRow, -1);
 950:         }
 951:         // 原来传 store.layer() —— 参数就是自己，bestFloor 永远停在 0。
 952:         // 现在 bestFloor 记录真实到达过的最高层（floorIndex 口径）。
 953:         store.raiseFloor(store.floorIndex());
 954: 
 955:         // 往下一层 = 变强一级。不显示 Lv. 数字，但变化本身要让玩家知道，
 956:         // 否则「越爬越强」这件事在界面上完全不可见。
 957:         // 提升封顶后就不再刷提示，免得每层都弹一句废话。
 958:         int eff = Math.min(store.floorIndex(), GBattle.LEVEL_BONUS_CAP);
 959:         if (eff >= 2) {
 960:             String extra = (eff == 2 || eff == 5) ? "，能量上限 +1" : "";
 961:             toast("⚡ 你变强了：伤害 +" + ((eff - 1) * 8) + "%" + extra);
 962:         }
 963:     }

 100:      *    但没考虑"整个平面都是特征方向"（单位阵倍数）。此时任意向量
 101:      *    都是特征向量，正确做法是给出两条坐标轴而不是空手而归。
 102:      */
 103:     public static Eigen eigen(Mat m) {
 104:         Eigen e = new Eigen();
 105:         e.trace = trace(m);
 106:         e.det = det(m);
 107:         e.disc = disc(m);
 108: 
 109:         double t = e.trace;
 110:         double d = e.disc;
 111: 
 112:         // 整个平面都是特征方向：A = kI
 113:         if (Math.abs(m.b) < 1e-9 && Math.abs(m.c) < 1e-9
 114:                 && Math.abs(m.a - m.d) < 1e-9) {
 115:             e.hasReal = true;
 116:             e.scalar = true;
 117:             e.l1 = m.a;
 118:             e.l2 = m.a;
 119:             e.v1 = new Vec(1, 0);
 120:             e.v2 = new Vec(0, 1);
 121:             return e;
 122:         }
 123: 
 124:         if (d < -1e-9) {
 125:             // 一对共轭复根：没有实的不动直线（会把平面转起来）
 126:             e.hasReal = false;
 127:             e.l1 = t / 2;
 128:             e.l2 = t / 2;
 129:             e.imag = Math.sqrt(-d) / 2;
 130:             return e;
 131:         }
 132: 
 133:         e.hasReal = true;
 134:         double sq = Math.sqrt(Math.max(0, d));
 135:         e.l1 = (t + sq) / 2;
 136:         e.l2 = (t - sq) / 2;
 137: 
 138:         e.v1 = eigenVec(m, e.l1);
 139:         // 重根只看一个方向；两不同根才找第二个
 140:         if (sq > 1e-9) e.v2 = eigenVec(m, e.l2);
 141:         else e.v2 = null;
 142:         return e;
 143:     }
 144: 
 145:     /**
 146:      * 求 λ 对应的单位特征向量。
 147:      * 解 (A − λI)v = 0。取法向量 (b, λ−a) 或 (λ−d, c)，哪个不退化用哪个。
 148:      */
 149:     public static Vec eigenVec(Mat m, double lambda) {
 150:         double bx = m.b, by = lambda - m.a;
 151:         if (Math.hypot(bx, by) > 1e-9) return new Vec(bx, by).norm();
 152: 
 153:         double cx = lambda - m.d, cy = m.c;
 154:         if (Math.hypot(cx, cy) > 1e-9) return new Vec(cx, cy).norm();
 155: 
 156:         return null;
 157:     }

 159:     /** 校验：M·v 是否等于 λ·v（给审计用，也让界面能自检）。 */
 160:     public static double residual(Mat m, Vec v, double lambda) {
 161:         Vec mv = apply(m, v.x, v.y);
 162:         return Math.hypot(mv.x - lambda * v.x, mv.y - lambda * v.y);
 163:     }

 210:                 D, D, e.l1, e.l2);

 495:         private void drawBasis(Canvas c, float cx, float cy, float scale) {
 496:             double lenI = Math.hypot(cur.a, cur.c);
 497:             double lenJ = Math.hypot(cur.b, cur.d);
 498:             if (lenI >= lenJ) {
 499:                 arrow(c, cx, cy, scale, 0, 0, cur.a, cur.c, pRed);
 500:                 arrow(c, cx, cy, scale, 0, 0, cur.b, cur.d, pTeal);
 501:             } else {
 502:                 arrow(c, cx, cy, scale, 0, 0, cur.b, cur.d, pTeal);
 503:                 arrow(c, cx, cy, scale, 0, 0, cur.a, cur.c, pRed);
 504:             }
 505:         }

 507:         private void arrow(Canvas c, float cx, float cy, float scale,
 508:                            double x1, double y1, double x2, double y2, Paint p) {
 509:             float sx1 = px(cx, scale, x1), sy1 = py(cy, scale, y1);
 510:             float sx2 = px(cx, scale, x2), sy2 = py(cy, scale, y2);
 511:             c.drawLine(sx1, sy1, sx2, sy2, p);
 512:             float dx = sx2 - sx1, dy = sy2 - sy1;
 513:             if (Math.hypot(dx, dy) < dp0(8)) return;
 514:             double ang = Math.atan2(dy, dx);
 515:             float hl = dp0(8);
 516:             Path head = new Path();
 517:             head.moveTo(sx2, sy2);
 518:             head.lineTo((float) (sx2 - hl * Math.cos(ang - 0.42)),
 519:                         (float) (sy2 - hl * Math.sin(ang - 0.42)));
 520:             head.lineTo((float) (sx2 - hl * Math.cos(ang + 0.42)),
 521:                         (float) (sy2 - hl * Math.sin(ang + 0.42)));
 522:             head.close();
 523:             Paint fp = new Paint(p);
 524:             fp.setStyle(Paint.Style.FILL);
 525:             c.drawPath(head, fp);
 526:         }

 532:          * 画不出金线，若图例还写着"λ 轴"，用户会以为自己漏看了。

 543:             } else {

 546:         }
