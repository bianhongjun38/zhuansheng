  19:  */

  22:     private static final String MAGIC = "ZSB1";
  23:     // Crockford Base32：去掉 I L O U，避免和 1 0 混淆
  24:     private static final String ALPHABET = "0123456789ABCDEFGHJKMNPQRSTVWXYZ";
  25: 
  26:     /** 把 exportJson() 的结果编码成进度码。 */
  27:     public static String encode(String json) {
  28:         try {
  29:             byte[] raw = json.getBytes("UTF-8");
  30:             ByteArrayOutputStream bo = new ByteArrayOutputStream();
  31:             DeflaterOutputStream d = new DeflaterOutputStream(bo, new Deflater(9), 8192);
  32:             d.write(raw);
  33:             d.finish();
  34:             d.close();
  35:             String b32 = base32(bo.toByteArray());
  36:             return MAGIC + "-" + group(b32) + "-" + checksum(b32);
  37:         } catch (Exception e) {
  38:             return null;
  39:         }
  40:     }

  42:     /** 解出 JSON。失败返回 null，不抛异常。 */
  43:     public static String decode(String code) {
  44:         try {
  45:             if (code == null) return null;
  46:             // 容忍用户粘贴时带了空格、换行、大小写混用
  47:             String s = code.trim().toUpperCase().replaceAll("[\\s\\-]", "");
  48:             if (!s.startsWith(MAGIC)) return null;
  49:             s = s.substring(MAGIC.length());
  50:             if (s.length() < 6) return null;
  51: 
  52:             String sum = s.substring(s.length() - 4);
  53:             String body = s.substring(0, s.length() - 4);
  54:             if (!checksum(body).equals(sum)) return null;   // 抄错了
  55:

  60:             ByteArrayOutputStream bo = new ByteArrayOutputStream();
  61:             byte[] buf = new byte[8192];
  62:             int n;
  63:             while ((n = in.read(buf)) > 0) bo.write(buf, 0, n);
  64:             in.close();

  72:     private static String base32(byte[] data) {
  73:         StringBuilder sb = new StringBuilder();
  74:         int buf = 0, bits = 0;
  75:         for (byte b : data) {
  76:             buf = (buf << 8) | (b & 0xFF);
  77:             bits += 8;
  78:             while (bits >= 5) {
  79:                 sb.append(ALPHABET.charAt((buf >> (bits - 5)) & 31));
  80:                 bits -= 5;
  81:             }
  82:         }
  83:         if (bits > 0) {
  84:             sb.append(ALPHABET.charAt((buf << (5 - bits)) & 31));
  85:         }
  86:         return sb.toString();
  87:     }

  89:     private static byte[] unBase32(String s) {
  90:         ByteArrayOutputStream bo = new ByteArrayOutputStream();
  91:         int buf = 0, bits = 0;
  92:         for (int i = 0; i < s.length(); i++) {
  93:             int v = ALPHABET.indexOf(s.charAt(i));
  94:             if (v < 0) return null;
  95:             buf = (buf << 5) | v;
  96:             bits += 5;
  97:             if (bits >= 8) {
  98:                 bo.write((buf >> (bits - 8)) & 0xFF);
  99:                 bits -= 8;

 101:         }

 112:         for (int i = 0; i < 4; i++) {
 113:             sb.append(ALPHABET.charAt((int) ((h >> (i * 5)) & 31)));
 114:         }
 115:         return sb.toString();

 289:     private void render() {
 290:         if (idx >= queue.size()) { finish(); return; }
 291:         final JSONObject w = queue.get(idx);
 292:         flipped = false;
 293: 
 294:         tvProgress.setText((idx + 1) + " / " + queue.size()
 295:                 + "   ✓" + known + "  ✕" + unknown);
 296:         tvWord.setText(w.optString("w", ""));
 297:         String ph = w.optString("ph", "");
 298:         tvPhon.setText(ph);
 299:         tvPhon.setVisibility(ph.isEmpty() ? View.GONE : View.VISIBLE);
 300:         // 背面先藏起来，逼自己先回忆
 301:         tvFace.setText("…");
 302:         tvFace.setTextColor(C_FG3);
 303:         tvFace.setTextSize(15);
 304:         tvExample.setText("");

 314:         answerBar.addView(see, lp);
 315:     }

 317:     private void flip() {
 318:         if (idx >= queue.size()) return;
 319:         JSONObject w = queue.get(idx);
 320:         flipped = true;
 321:         tvFace.setTextColor(C_FG);
 322:         tvFace.setTextSize(19);
 323:         String pos = w.optString("pos", "");
 324:         String cn = w.optString("cn", "");
 325:         tvFace.setText(pos.isEmpty() ? cn : (pos + "  " + cn));
 326: 
 327:         String ex = w.optString("ex", "");
 328:         String excn = w.optString("exCn", "");
 329:         if (!ex.isEmpty()) {
 330:             StringBuilder sb = new StringBuilder();
 331:             sb.append(ex);
 332:             if (!excn.isEmpty()) sb.append("\n").append(excn);
 333:             tvExample.setText(sb.toString());
 334:         } else {
 335:             String phr = w.optString("phr", "");
 336:             tvExample.setText(phr);
 337:         }
 338: 
 339:         // 翻面后换成两个判断按钮：认识 / 不认识
 340:         answerBar.removeAllViews();
 341:         bottomBar.removeAllViews();
 342: 
 343:         TextView no = mkBtn("不认识", C_RED, new View.OnClickListener() {
 344:             @Override public void onClick(View v) { mark(false); }
 345:         });
 346:         TextView yes = mkBtn("认识", C_TEAL, new View.OnClickListener() {
 347:             @Override public void onClick(View v) { mark(true); }
 348:         });
 349:         LinearLayout.LayoutParams lp1 = new LinearLayout.LayoutParams(
 350:                 0, ViewGroup.LayoutParams.WRAP_CONTENT, 1f);
 351:         lp1.setMargins(0, 0, dp(5), 0);
 352:         LinearLayout.LayoutParams lp2 = new LinearLayout.LayoutParams(
 353:                 0, ViewGroup.LayoutParams.WRAP_CONTENT, 1f);
 354:         lp2.setMargins(dp(5), 0, 0, 0);
 355:         bottomBar.addView(no, lp1);
 356:         bottomBar.addView(yes, lp2);
 357:     }

 359:     private void mark(boolean ok) {
 360:         if (idx >= queue.size()) return;
 361:         JSONObject w = queue.get(idx);
 362:         if (ok) {
 363:             known++;
 364:         } else {
 365:             unknown++;
 366:             again.add(w);      // 进"再背一遍"，本组结束前必须再过
 367:         }
 368:         idx++;
 369:         if (idx >= queue.size()) {
 370:             if (!again.isEmpty()) {
 371:                 // 把不认识的放回队列尾部，清空 again 重新计
 372:                 int n = again.size();
 373:                 queue.addAll(again);
 374:                 again.clear();
 375:                 render();
 376:                 Toast("还有 " + n + " 个要再过一遍");
 377:                 return;
 378:             }
 379:             finish();
 380:             return;
 381:         }
 382:         render();
 383:     }
