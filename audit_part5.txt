 675:     private void addRecallTop() {
 676:         if (idx < 0 || idx >= plan.days.size()) return;
 677:         Plan.Day d = plan.days.get(idx);
 678: 
 679:         // 候选章：① 今天到期要复习的（优先，这是记忆曲线的正事）
 680:         //         ② 今天新学的（保证每天都有卡，不然头几天打开是空的）
 681:         // 都按出现顺序去重；没有答案的章直接跳过（宁缺勿编）。
 682:         java.util.LinkedHashSet<String> seen = new java.util.LinkedHashSet<>();
 683:         for (Plan.Review r : d.reviews) {
 684:             if (plan.recall.containsKey(r.chapterId)) seen.add(r.chapterId);
 685:         }
 686:         int fromReview = seen.size();
 687:         if (seen.isEmpty()) {
 688:             for (Plan.Task t : d.tasks) {
 689:                 if (plan.recall.containsKey(t.chapterId)) seen.add(t.chapterId);
 690:             }
 691:         }
 692:         // ③ 兜底：越靠后越是整天刷真题/错题，没有"新学的章"。
 693:         // 但"每日置顶"不能因为今天是综合练习就空着——那就退回到
 694:         // 最近学过的那几章，让用户每天都有东西扫一眼。
 695:         if (seen.isEmpty()) {
 696:             for (int i = idx; i >= 0 && seen.isEmpty(); i--) {
 697:                 for (Plan.Task t : plan.days.get(i).tasks) {
 698:                     if (plan.recall.containsKey(t.chapterId)) seen.add(t.chapterId);
 699:                 }
 700:             }
 701:         }
 702:         if (seen.isEmpty()) return;
 703: 
 704:         String sub;
 705:         if (fromReview > 0) {
 706:             sub = "今天到期的 " + fromReview + " 章 · 答案直接摊开，不用默写";

 708:             sub = "今天新学的 " + seen.size() + " 章 · 先背住这些结论";
 709:         } else {
 710:             sub = "最近学过的 " + seen.size() + " 章 · 综合练习日再看一遍";
 711:         }
 712: 
 713:         LinearLayout card = card("今日必背 · " + seen.size() + " 章", sub);
 714: 
 715:         int rank = 0;

 719:

 535:     /** 按一个键：插到光标处（或替换选区），然后刷新。 */
 536:     private void applyKey(String frag) {
 537:         if (etExpr == null) return;
 538: 
 539:         // 翻页：不插入任何内容，只换一张键盘
 540:         if (ExprInput.SWITCH.equals(frag)) {
 541:             keyPage = 1;
 542:             buildKeysInto(keyWrap, 6);
 543:             return;
 544:         }
 545:         if (ExprInput.BACK_TO_NUM.equals(frag)) {
 546:             keyPage = 0;
 547:             buildKeysInto(keyWrap, 6);
 548:             return;
 549:         }
 550:         int a = etExpr.getSelectionStart();
 551:         int b = etExpr.getSelectionEnd();
 552:         // Android 的 getSelectionStart 在极端情况下会返回 -1
 553:         if (a < 0) a = etExpr.getText().length();
 554:         if (b < 0) b = a;
 555: 
 556:         ExprInput.R r = ExprInput.insert(etExpr.getText().toString(), a, b, frag);
 557:         etExpr.setText(r.text);
 558:         int pos = Math.max(0, Math.min(r.cursor, etExpr.getText().length()));
 559:         etExpr.setSelection(pos);
 560: 
 561:         // 括号没闭上就补一下，省得用户数括号
 562:         int un = ExprInput.unclosedParens(r.text);
 563:         if (un != 0) updateKeyHint(un);
 564:         else updateKeyHint(0);
 565:         if (!"\b".equals(frag)) {
 566:             // 插入后立刻重画，边写边看
 567:             exprText = etExpr.getText().toString();
 568:             draw();
 569:         }
 570:     }

 781:     private void updateInfo() {
 782:         if (tvPlotInfo == null || plot == null) return;
 783:         Expr.Node f = plot.expr();
 784:         if (f == null) { tvPlotInfo.setText("还没输入函数"); return; }
 785: 
 786:         StringBuilder b = new StringBuilder();
 787:         b.append("f(x) = ").append(exprText);
 788:         Expr.Node d = plot.deriv();
 789:         if (d != null) b.append("\nf′(x) = ").append(Expr.toText(d));
 790: 
 791:         List<Double> rs = PlotGeom.roots(f, plot.window(), 600);
 792:         if (!rs.isEmpty()) {
 793:             b.append("\n零点：");
 794:             for (int i = 0; i < rs.size() && i < 5; i++) {
 795:                 if (i > 0) b.append("，");
 796:                 b.append("x=").append(Expr.fmtNum(rs.get(i)));
 797:             }
 798:             if (rs.size() > 5) b.append(" 等 ").append(rs.size()).append(" 个");
 799:         } else {
 800:             b.append("\n零点：本窗口内没有");
 801:         }
 802: 
 803:         Double tx = plot.tangentX();
 804:         if (tx != null && plot.isShowTangent()) {
 805:             double y = PlotGeom.safeEval(f, tx);
 806:             double k = PlotGeom.slope(f, tx);
 807:             b.append("\n点 x=").append(Expr.fmtNum(tx))
 808:                     .append("：f(x)=").append(Expr.fmtNum(y))
 809:                     .append("，f′=").append(Expr.fmtNum(k));
 810:             if (PlotGeom.isFinite(k) && Math.abs(k) > 1e-9) {
 811:                 b.append("，切线 y=").append(Expr.fmtNum(k)).append("x");
 812:                 double c = y - k * tx;
 813:                 if (c >= 0) b.append("+").append(Expr.fmtNum(c));
 814:                 else b.append("−").append(Expr.fmtNum(-c));
 815:             } else if (PlotGeom.isFinite(k)) {
 816:                 b.append("，切线水平 y=").append(Expr.fmtNum(y));
 817:             }
 818:         }
 819:         tvPlotInfo.setTextColor(C_FG2);
 820:         tvPlotInfo.setText(b.toString());
 821:     }

1071:     /**
1072:      * 判分。不比对文本，比对"数值上是不是同一个函数"：
1073:      * 学生可能写成 (2)·x 或 x·2 或 2 x，字面不同但都对。
1074:      * 做法：在 5 个点上比较他的表达式与标准导数；全部接近才算对。
1075:      * 这样既接受等价写法，也不会把 x 和 2x 判成一样。
1076:      */
1077:     private String judgeDeriv(Expr.Node f, String input) {
1078:         if (input == null || input.trim().isEmpty()) return "还没写";
1079:         Expr.Node mine;
1080:         try {
1081:             mine = Expr.parse(input);
1082:         } catch (Expr.ExprError e) {
1083:             return "表达式看不懂：" + e.getMessage();
1084:         }
1085:         Expr.Node ref = Expr.diff(f);
1086:         String refText = Expr.toText(ref);
1087: 
1088:         double[] xs = {-2.3, -1.1, -0.4, 0.6, 1.4, 2.2};
1089:         int tested = 0;
1090:         for (double x : xs) {
1091:             double a = Expr.eval(ref, x);
1092:             double b = Expr.eval(mine, x);
1093:             if (!PlotGeom.isFinite(a) || !PlotGeom.isFinite(b)) continue;
1094:             if (Math.abs(a) > 1e6 || Math.abs(b) > 1e6) continue;
1095:             tested++;
1096:             double scale = Math.max(1, Math.max(Math.abs(a), Math.abs(b)));
1097:             if (Math.abs(a - b) > 1e-6 * scale) {
1098:                 return "✗ 不对。\n正确答案：" + refText
1099:                         + "\n（你在 x=" + Expr.fmtNum(x) + " 处得到 "
1100:                         + Expr.fmtNum(b) + "，正确是 " + Expr.fmtNum(a) + "）";
1101:             }
1102:         }
1103:         if (tested == 0) {
1104:             return "✗ 无法判定：你的表达式和标准答案没有公共定义域。\n正确答案：" + refText;
1105:         }
1106:         if (Expr.toText(mine).equals(refText)) {
1107:             return "✓ 正确，而且写法和标准答案一致：" + refText;
1108:         }
1109:         return "✓ 正确（写法不同但等价）。\n你写的：" + Expr.toText(mine)
1110:                 + "\n标准写法：" + refText;
1111:     }
